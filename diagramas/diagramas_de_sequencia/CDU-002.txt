title CDU 002 - Registrar Objeto

participant IndexUI
participant RegistrarObjetoUI
participant ObjetoController
participant ObjetoDAO
participant Objeto

IndexUI -> RegistrarObjetoUI: main()

RegistrarObjetoUI -> ObjetoController: inserir(descricao, data_hora_encontrado, local_encontrado) : Objeto

ObjetoController -> Objeto: __new__(descricao, data_hora_encontrado, local_encontrado) : Objeto

ObjetoController -> ObjetoDAO: inserir(objeto) : Objeto

ObjetoDAO -> ObjetoDAO: abrir()

activate ObjetoDAO
  loop for each dados_objeto
    ObjetoDAO -> Objeto: from_json(dados_objeto): Objeto
  end
deactivate ObjetoDAO

ObjetoDAO -> ObjetoDAO: salvar()

activate ObjetoDAO
  loop for each objeto
    ObjetoDAO -> Objeto: objeto.to_json() : dict[str, Any]
  end
deactivate ObjetoDAO
